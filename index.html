-- Incremental load: new orders only
WITH staged AS (
  SELECT
    order_id,
    customer_id,
    PARSE_JSON(payload):amount::NUMBER AS amount,
    CURRENT_TIMESTAMP() AS loaded_at
  FROM raw.orders_stream
  WHERE _metadata_action = 'INSERT'
)
MERGE INTO analytics.fact_orders t
USING staged s ON t.order_id = s.order_id
WHEN NOT MATCHED THEN INSERT
  (order_id, customer_id, amount, loaded_at)
VALUES
  (s.order_id, s.customer_id, s.amount, s.loaded_at);
